  <h1>Buscaminas</h1>

<script>
(function () {
  var COLS = 9, ROWS = 9, MINES = 10;
  var mine, open, flag, near;
  var over, won, started, flags, seconds, ticker;

  var grid = document.getElementById('grid');
  var msg = document.getElementById('msg');
  var minesLeft = document.getElementById('minesLeft');
  var timer = document.getElementById('timer');
  var resetBtn = document.getElementById('reset');

  function idx(r, c) { return r * COLS + c; }
  function inBounds(r, c) { return r >= 0 && r < ROWS && c >= 0 && c < COLS; }

  function neighbors(r, c) {
    var out = [];
    for (var dr = -1; dr <= 1; dr++) {
      for (var dc = -1; dc <= 1; dc++) {
        if (dr === 0 && dc === 0) continue;
        if (inBounds(r + dr, c + dc)) out.push([r + dr, c + dc]);
      }
    }
    return out;
  }

  function reset() {
    var total = ROWS * COLS;
    mine = new Array(total).fill(false);
    open = new Array(total).fill(false);
    flag = new Array(total).fill(false);
    near = new Array(total).fill(0);
    over = false; won = false; started = false;
    flags = 0; seconds = 0;
    clearInterval(ticker); ticker = null;
    resetBtn.textContent = '🙂';
    msg.textContent = 'Clic para descubrir · clic derecho para marcar';
    msg.className = '';
    timer.textContent = '⏱ 0';
    updateMines();
    draw();
  }

  // Las minas se colocan tras el primer clic para que nunca sea perdedor.
  function placeMines(safeR, safeC) {
    var forbidden = {};
    forbidden[idx(safeR, safeC)] = true;
    neighbors(safeR, safeC).forEach(function (p) { forbidden[idx(p[0], p[1])] = true; });

    var placed = 0;
    while (placed < MINES) {
      var i = Math.floor(Math.random() * ROWS * COLS);
      if (mine[i] || forbidden[i]) continue;
      mine[i] = true;
      placed++;
    }
    for (var r = 0; r < ROWS; r++) {
      for (var c = 0; c < COLS; c++) {
        var count = 0;
        neighbors(r, c).forEach(function (p) { if (mine[idx(p[0], p[1])]) count++; });
        near[idx(r, c)] = count;
      }
    }
  }

  function startTimer() {
    ticker = setInterval(function () {
      seconds++;
      timer.textContent = '⏱ ' + seconds;
    }, 1000);
  }

  function updateMines() {
    minesLeft.textContent = '💣 ' + (MINES - flags);
  }

  function reveal(r, c) {
    var i = idx(r, c);
    if (open[i] || flag[i]) return;
    open[i] = true;
    if (near[i] === 0 && !mine[i]) {
      neighbors(r, c).forEach(function (p) { reveal(p[0], p[1]); });
    }
  }

  function checkWin() {
    for (var i = 0; i < ROWS * COLS; i++) {
      if (!mine[i] && !open[i]) return false;
    }
    return true;
  }

  function endGame(victory) {
    over = true; won = victory;
    clearInterval(ticker); ticker = null;
    resetBtn.textContent = victory ? '😎' : '💀';
    msg.textContent = victory
      ? '¡Ganaste en ' + seconds + ' segundos!'
      : 'Pisaste una mina. Pulsa la cara para reintentar.';
    msg.className = victory ? 'win' : 'lose';
  }

  function onLeft(r, c) {
    if (over) return;
    var i = idx(r, c);
    if (flag[i] || open[i]) return;

    if (!started) {
      started = true;
      placeMines(r, c);
      startTimer();
    }

    if (mine[i]) {
      for (var k = 0; k < ROWS * COLS; k++) { if (mine[k]) open[k] = true; }
      open[i] = true;
      draw(i);
      endGame(false);
      return;
    }

    reveal(r, c);
    draw();
    if (checkWin()) endGame(true);
  }

  function onRight(r, c) {
    if (over) return;
    var i = idx(r, c);
    if (open[i]) return;
    flag[i] = !flag[i];
    flags += flag[i] ? 1 : -1;
    updateMines();
    draw();
  }

  function draw(boomIndex) {
    grid.textContent = '';
    for (var r = 0; r < ROWS; r++) {
      for (var c = 0; c < COLS; c++) {
        var i = idx(r, c);
        var el = document.createElement('div');
        el.className = 'c';

        if (open[i]) {
          el.className += ' open';
          if (mine[i]) {
            el.textContent = '💣';
            if (i === boomIndex) el.className += ' boom';
          } else if (near[i] > 0) {
            el.textContent = String(near[i]);
            el.className += ' n' + near[i];
          }
        } else if (flag[i]) {
          el.textContent = '🚩';
        }

        (function (rr, cc) {
          el.addEventListener('click', function () { onLeft(rr, cc); });
          el.addEventListener('contextmenu', function (e) {
            e.preventDefault();
            onRight(rr, cc);
          });
        })(r, c);

        grid.appendChild(el);
      }
    }
  }

  resetBtn.addEventListener('click', reset);
  reset();
})();
